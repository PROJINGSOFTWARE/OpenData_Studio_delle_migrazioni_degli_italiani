import pandas as pd
# PATH: cartella dove si trovano i dataset originali
file_path = "C:\\Users\\Sveva\\Desktop\\OpenData_assignment\\territorio_destinazione_coordinate.csv"
mi_path = "C:\\Users\\Sveva\\Desktop\\OpenData_assignment\\Migrazioni_URI\\migrazione.csv"
tr_path = "C:\\Users\\Sveva\\Desktop\\OpenData_assignment\\Migrazioni_URI\\territorio.csv"
#######
#######
td = pd.read_csv(file_path, sep=",", index_col=None)
######
######
migrazione = td[["Territorio di origine", "Territorio di destinazione", "Paese di cittadinanza", "Sesso", "anno", "Tipo di trasferimento", "valore"]]
territorio = td[["Cod_destinazione", "Territorio di destinazione", "latitudine_d", "longitudine_d"]]
migrazione.columns = ["origine", "destinazione", "cittadinanza", "sesso", "anno", "trasferimento", "valore"]
territorio.columns = ["codice", "nome", "latitudine", "longitudine"]
migrazione = migrazione.drop_duplicates(ignore_index=True)
territorio = territorio.drop_duplicates(ignore_index=True)
migrazione.to_csv(mi_path, encoding="utf-8", index=None)
territorio.to_csv(tr_path, encoding="utf-8", index=None)
