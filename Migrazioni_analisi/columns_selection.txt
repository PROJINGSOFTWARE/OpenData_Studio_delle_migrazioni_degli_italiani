def kept_columns_csv(raw_csv_namefile, kept_columns, new_name_of_columns):
    path = base_path + raw_csv_namefile
    df = pd.read_csv(path, sep=",")
    df = df[kept_columns]
    df.columns = new_name_of_columns
    df = df.drop_duplicates(ignore_index=True)
    new_path = new_base_path + raw_csv_namefile
    csv_df = df.to_csv(new_path + "", encoding="utf-8", index=None)
