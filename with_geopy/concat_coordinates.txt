import pandas as pd
base_path = "C:\\Users\\Sveva\\Desktop\\OpenData_assignment\\with_geopy\\"
cc = "country_coordinates.csv"

cc_path = base_path + cc

mondo = {Cod_Paese:"X1033",Latitudine:"",Longitude:"" }

pd.concat([cc_df, mondo],ignore_index=True)
it_c_df = pd.read_csv(it_c_path, sep=",", encoding="utf-8", index_col=None)
